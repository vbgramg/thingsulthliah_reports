How Supabase Free works:

SQL Setup Query (Run in Supabase SQL Editor): 
create table if not exists village_demands (
  id uuid default gen_random_uuid() primary key,
  created_at timestamptz default now() not null,
  village text not null,
  month text not null,
  file_name text not null,
  file_url text not null,
  file_size bigint,
  submitted_by text,
  remarks text
);
insert into storage.buckets (id, name, public) values ('village-demands', 'village-demands', true) on conflict do nothing;
create policy "Allow Public Uploads" on storage.objects for insert with check (bucket_id = 'village-demands');
create policy "Allow Public Reads" on storage.objects for select using (bucket_id = 'village-demands');
alter table village_demands enable row level security;
create policy "Allow Public Select" on village_demands for select using (true);
create policy "Allow Public Insert" on village_demands for insert with check (true);
create policy "Allow Public Delete" on village_demands for delete using (true);
